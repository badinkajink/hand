\begin{figure}[t]

    \coordinate (J1) at (0,-20.75);

    \coordinate (E2) at (0,-78.66); % -41.50 - 37.16

    \draw[rounded corners=2.5mm,fill=red!10,draw=red!65!black,line width=0.75pt]
      (-10.55,0) rectangle (10.55,-33.45);
    \draw[rounded corners=2.5mm,fill=blue!10,draw=blue!65!black,line width=0.75pt]
      (-9.55,-20.75) rectangle (11.55,-54.20);

    % Joint centers.
    \node[joint,label={[lbl]left:Yaw}] at (J0) {$\odot$};
    \node[joint,label={[lbl]left:MCP}] at (J1) {$\otimes$};
    \node[joint,label={[lbl]left:PIP}] at (J2) {$\otimes$};

    % Link labels.
    \node[lbl,red!70!black,anchor=west] at (14,-9) {yaw link};
    \node[lbl,blue!70!black,anchor=west] at (14,-31) {middle link};
    \node[lbl,anchor=west] at (14,-63) {distal link / fingertip};

    % Joint-center spacing: 33.45 - 12.70 = 20.75 mm.
    \draw[dim] (-15,0) -- (-15,-20.75)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {20.75 mm};
    \draw[dim] (-15,-20.75) -- (-15,-41.50)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {20.75 mm};

    % Axis-to-distal-surface link lengths.
    \draw[dim] (31,0) -- (31,-33.45)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {33.45 mm};
    \draw[dim] (37,-20.75) -- (37,-54.20)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {33.45 mm};
    \draw[dim] (31,-41.50) -- (31,-78.66)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {37.16 mm};

    % Parent-link overhang past the child joint axis.
    \draw[dim] (24,-20.75) -- (24,-33.45)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {12.70 mm};
    \draw[dim] (24,-41.50) -- (24,-54.20)
      node[midway,fill=white,inner sep=1pt,rotate=90,lbl] {12.70 mm};

    % Joint axis to phalange outer surface (half-width).
    \draw[dim] (0,5.5) -- (10.55,5.5)
      node[midway,fill=white,inner sep=1pt,lbl] {10.55 mm};
    \draw[ext] (0,2.5) -- (0,7.8);
    \draw[ext] (10.55,2.5) -- (10.55,7.8);

    % Local MuJoCo axes / zero pose.
    \begin{scope}[shift={(55,-4)}]
      \draw[axis] (0,0) -- (11,0) node[right,lbl] {$+y$ (MCP/PIP axis)};

      \node[lbl,anchor=west] at (-2,5) {$+x$ yaw axis: $\odot$};
    \end{scope}

    % ROM annotation.
    \node[draw,rounded corners=1.5mm,align=left,anchor=north west,font=\small,

      $q_{\mathrm{PIP}}\in[-18^\circ,\,92^\circ]$\\[1mm]
      $q=0$: straight / vertical\\
      positive: abduction or flexion
    };

  \label{fig:real-hand-kinematics}
